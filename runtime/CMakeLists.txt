cmake_minimum_required(VERSION 3.20)
project(strata VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
option(BUILD_SHARED_LIBS "Build shared library for FFI embedding" ON)
option(STRATA_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(STRATA_DIRECTML "Enable a DirectML-enabled Windows ORT package" OFF)
set(ONNXRUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "ONNX Runtime SDK")
unset(ONNXRUNTIME_INCLUDE CACHE)
unset(ONNXRUNTIME_LIBRARY CACHE)
find_path(ONNXRUNTIME_INCLUDE onnxruntime_c_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(OpenEXR 3 REQUIRED CONFIG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wshadow -Wconversion -ffp-contract=off)
endif()
if(STRATA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(libstrata src/backend/runtime.c src/core/image.c src/core/exr.c src/core/depth.c src/core/contract.c src/core/deep.c src/core/position.c src/core/camera_io.c)
set_target_properties(libstrata PROPERTIES OUTPUT_NAME strata VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(libstrata PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(libstrata PRIVATE "${STRATA_ROOT}/contract")
target_include_directories(libstrata SYSTEM PRIVATE "${ONNXRUNTIME_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb")
target_link_libraries(libstrata PRIVATE "${ONNXRUNTIME_LIBRARY}" OpenEXR::OpenEXRCore PkgConfig::JSONC)
if(STRATA_DIRECTML)
  if(NOT WIN32)
    message(FATAL_ERROR "DirectML is Windows-only")
  endif()
  target_compile_definitions(libstrata PRIVATE STRATA_DIRECTML)
endif()
if(WIN32)
  set_target_properties(libstrata PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
else()
  target_link_libraries(libstrata PRIVATE m)
endif()
add_executable(strata src/cli/cli.c)
target_link_libraries(strata PRIVATE libstrata)
if(NOT WIN32)
  target_link_libraries(strata PRIVATE m)
endif()
get_filename_component(ORT_LIBRARY_DIR "${ONNXRUNTIME_LIBRARY}" DIRECTORY)
set_target_properties(libstrata PROPERTIES INSTALL_RPATH "${ORT_LIBRARY_DIR}")
set_target_properties(strata PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR};${ORT_LIBRARY_DIR}")
install(TARGETS libstrata strata LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/strata.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(CLANG_FORMAT clang-format REQUIRED)
  set(STRATA_TEST_PYTHON "${Python3_EXECUTABLE}" CACHE FILEPATH "Python with numpy, OpenEXR and Pillow")
  set(STRATA_MODEL "${STRATA_ROOT}/models/depthpro2-h.onnx" CACHE FILEPATH "Exported H+ model")
  set(STRATA_PARITY_MANIFEST "${STRATA_ROOT}/build/parity.json" CACHE FILEPATH "Parity image/reference manifest")
  set(STRATA_REFERENCE_DIR "${STRATA_ROOT}/build/reference" CACHE PATH "PyTorch oracle results")
  add_executable(strata_position_test tests/position.c)
  target_link_libraries(strata_position_test PRIVATE libstrata)
  add_test(NAME position_plane COMMAND strata_position_test)
  add_executable(strata_deep_test tests/deep.c)
  target_link_libraries(strata_deep_test PRIVATE libstrata)
  add_test(NAME deep_memory COMMAND strata_deep_test)
  add_executable(strata_session_fixture tests/session.c)
  target_link_libraries(strata_session_fixture PRIVATE libstrata)
  add_executable(strata_io_fixture tests/io.c)
  target_link_libraries(strata_io_fixture PRIVATE libstrata)
  add_test(NAME deep_readback COMMAND "${STRATA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/deep_readback.py" $<TARGET_FILE:strata_io_fixture> "${CMAKE_CURRENT_BINARY_DIR}/io-test")
  add_test(NAME parity COMMAND "${STRATA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.py" --cli $<TARGET_FILE:strata> --model "${STRATA_MODEL}" --manifest "${STRATA_PARITY_MANIFEST}" --reference-dir "${STRATA_REFERENCE_DIR}" --out "${CMAKE_CURRENT_BINARY_DIR}/parity")
  set_tests_properties(deep_readback parity PROPERTIES ENVIRONMENT "PYTHONPATH=" RUN_SERIAL TRUE)
  set_tests_properties(parity PROPERTIES LABELS gpu)
  add_test(NAME commit_rules COMMAND "${STRATA_TEST_PYTHON}" "${STRATA_ROOT}/tools/check_commit_rules.py")
  add_test(NAME layer_rules COMMAND "${STRATA_TEST_PYTHON}" "${STRATA_ROOT}/tools/check_layer_rules.py")
  add_test(NAME layers COMMAND bash "${STRATA_ROOT}/tools/check_layers.sh")
  add_test(NAME comments COMMAND bash "${STRATA_ROOT}/tools/check_comments.sh")
  add_test(NAME format COMMAND bash "${STRATA_ROOT}/tools/check_format.sh")
  add_test(NAME contract COMMAND "${STRATA_TEST_PYTHON}" "${STRATA_ROOT}/tools/check_contract.py" $<TARGET_FILE:strata> "${CMAKE_CURRENT_BINARY_DIR}/contract-test" $<TARGET_FILE:strata_session_fixture>)
  add_test(NAME cli_help COMMAND strata --help)
  set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "strata depth --model")
endif()
