cmake_minimum_required(VERSION 3.20)
project(strata VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
include(CTest)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
option(BUILD_SHARED_LIBS "Build shared library for FFI embedding" ON)
option(STRATA_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(ONNXRUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime" CACHE PATH "ONNX Runtime SDK")
unset(ONNXRUNTIME_INCLUDE CACHE)
unset(ONNXRUNTIME_LIBRARY CACHE)
find_path(ONNXRUNTIME_INCLUDE onnxruntime_c_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ONNXRUNTIME_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(OpenEXR 3 REQUIRED CONFIG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wshadow -Wconversion -ffp-contract=off)
if(STRATA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(libstrata src/backend/runtime.c src/core/image.c src/core/exr.c src/core/depth.c src/core/contract.c src/core/deep.c src/core/position.c src/core/camera_io.c src/core/paths.c)
set_target_properties(libstrata PROPERTIES OUTPUT_NAME strata VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(libstrata PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(libstrata PRIVATE "${STRATA_ROOT}/contract")
target_include_directories(libstrata SYSTEM PRIVATE "${ONNXRUNTIME_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/stb")
target_link_libraries(libstrata PRIVATE "${ONNXRUNTIME_LIBRARY}" OpenEXR::OpenEXRCore PkgConfig::JSONC m ${CMAKE_DL_LIBS})
add_executable(strata src/cli/cli.c)
target_link_libraries(strata PRIVATE libstrata m)
add_executable(strata_bench src/bench/bench.c)
target_link_libraries(strata_bench PRIVATE libstrata m)
get_filename_component(ORT_LIBRARY_DIR "${ONNXRUNTIME_LIBRARY}" DIRECTORY)
set_target_properties(libstrata PROPERTIES INSTALL_RPATH "${ORT_LIBRARY_DIR}")
set_target_properties(strata strata_bench PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR};${ORT_LIBRARY_DIR}")
install(TARGETS libstrata strata strata_bench LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/strata.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(CLANG_FORMAT clang-format REQUIRED)
  set(STRATA_TEST_PYTHON "${Python3_EXECUTABLE}" CACHE FILEPATH "Python with numpy, OpenEXR, OpenImageIO and Pillow")
  set(STRATA_MODEL "${STRATA_ROOT}/models/depthpro2-h.onnx" CACHE FILEPATH "Exported H+ model")
  foreach(name position deep)
    add_executable(strata_${name}_test tests/${name}.c)
    target_link_libraries(strata_${name}_test PRIVATE libstrata m)
  endforeach()
  add_test(NAME position_plane COMMAND strata_position_test)
  add_test(NAME deep_memory COMMAND strata_deep_test)
  add_executable(strata_session_fixture tests/session.c)
  find_package(Threads REQUIRED)
  target_link_libraries(strata_session_fixture PRIVATE libstrata m Threads::Threads)
  add_executable(strata_srgb_fixture tests/srgb.c)
  target_include_directories(strata_srgb_fixture PRIVATE "${STRATA_ROOT}/contract")
  target_link_libraries(strata_srgb_fixture PRIVATE libstrata)
  add_executable(strata_io_fixture tests/io.c)
  target_link_libraries(strata_io_fixture PRIVATE libstrata m)
  add_test(NAME deep_readback COMMAND "${STRATA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/deep_readback.py" $<TARGET_FILE:strata_io_fixture> "${CMAKE_CURRENT_BINARY_DIR}/io-test")
  add_test(NAME cli_paths COMMAND "${STRATA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_paths.py" $<TARGET_FILE:strata> "${CMAKE_CURRENT_BINARY_DIR}/cli-paths-test")
  add_test(NAME parity_mask COMMAND "${STRATA_TEST_PYTHON}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
  set(parity_args --cli $<TARGET_FILE:strata> --model "${STRATA_MODEL}" --cases "${CMAKE_CURRENT_SOURCE_DIR}/tests/parity_cases.json" --out "${CMAKE_CURRENT_BINARY_DIR}/parity")
  set(gpu_environment "PYTHONPATH=")
  if(STRATA_SANITIZE)
    list(APPEND parity_args --lsan-suppression libcuda.so)
    set(gpu_environment "PYTHONPATH=;ASAN_OPTIONS=detect_leaks=1:halt_on_error=1:protect_shadow_gap=0;LSAN_OPTIONS=suppressions=${CMAKE_CURRENT_SOURCE_DIR}/tests/lsan.supp:print_suppressions=1")
    add_executable(strata_leak_canary tests/leak_canary.c)
    target_link_libraries(strata_leak_canary PRIVATE libstrata)
    add_test(NAME leak_canary COMMAND strata_leak_canary)
    set_tests_properties(leak_canary PROPERTIES ENVIRONMENT "${gpu_environment}" PASS_REGULAR_EXPRESSION "detected memory leaks.*strata_depth_to_deep")
  endif()
  add_test(NAME parity COMMAND "${STRATA_TEST_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.py" ${parity_args})
  set_tests_properties(deep_readback PROPERTIES ENVIRONMENT "PYTHONPATH=" RUN_SERIAL TRUE)
  set_tests_properties(parity_mask cli_paths PROPERTIES ENVIRONMENT "PYTHONPATH=")
  set_tests_properties(parity PROPERTIES ENVIRONMENT "${gpu_environment}" RUN_SERIAL TRUE LABELS gpu SKIP_RETURN_CODE 77)
  foreach(check commit_rules rule_cases)
    add_test(NAME ${check} COMMAND "${STRATA_TEST_PYTHON}" "${STRATA_ROOT}/tools/check_${check}.py")
  endforeach()
  foreach(check layers comments format decisions)
    add_test(NAME ${check} COMMAND bash "${STRATA_ROOT}/tools/check_${check}.sh")
  endforeach()
  add_test(NAME contract COMMAND "${STRATA_TEST_PYTHON}" "${STRATA_ROOT}/tools/check_contract.py" $<TARGET_FILE:strata> "${CMAKE_CURRENT_BINARY_DIR}/contract-test" $<TARGET_FILE:strata_session_fixture> $<TARGET_FILE:strata_srgb_fixture>)
  set_tests_properties(contract PROPERTIES ENVIRONMENT "${gpu_environment}")
  add_test(NAME cli_help COMMAND strata --help)
  set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "strata depth --model")
endif()
